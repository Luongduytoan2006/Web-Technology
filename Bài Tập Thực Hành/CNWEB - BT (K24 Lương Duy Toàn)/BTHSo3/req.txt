-- 1. Tạo cơ sở dữ liệu (nếu chưa có) và chọn CSDL (tên database tổng thể là DULIEU)
    CREATE DATABASE IF NOT EXISTS DULIEU
    CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

    USE DULIEU;

    -- 2. Tạo bảng admin (Tài khoản quản trị)
    CREATE TABLE IF NOT EXISTS admin (
        username VARCHAR(50) PRIMARY KEY,
        pass VARCHAR(255) NOT NULL
    );

    -- 3. Tạo bảng phongban (Cần tạo trước để nhanvien tham chiếu khóa ngoại)
    CREATE TABLE IF NOT EXISTS phongban (
        IDPB VARCHAR(10) PRIMARY KEY,
        Tenpb VARCHAR(100) NOT NULL,
        Mota TEXT
    );

    -- 4. Tạo bảng nhanvien (Liên kết với phongban qua IDPB)
    CREATE TABLE IF NOT EXISTS nhanvien (
        IDNV VARCHAR(10) PRIMARY KEY,
        Hoten VARCHAR(100) NOT NULL,
        IDPB VARCHAR(10),
        Diachi VARCHAR(255),
        CONSTRAINT fk_nhanvien_phongban
            FOREIGN KEY (IDPB) REFERENCES phongban(IDPB)
            ON DELETE SET NULL
            ON UPDATE CASCADE
    );
